const AGENT_ID_ADDR = "0x60B9262456dE011F7d15a54CdD8c48bAaa8B022F";

function toast(text, type = "info") {
  const box = document.getElementById("globalToast");
  const colors = {
    info: { border: "#f0b90b", color: "#eaecef", bg: "#15191f" },
    success: { border: "#0ecb81", color: "#0ecb81", bg: "rgba(14,203,129,.1)" },
    error: { border: "#f6465d", color: "#f6465d", bg: "rgba(246,70,93,.1)" }
  };
  const c = colors[type] || colors.info;
  box.style.display = "block";
  box.style.borderColor = c.border;
  box.style.color = c.color;
  box.style.background = c.bg;
  box.innerHTML = text;
  box.scrollIntoView({ behavior: "smooth", block: "nearest" });
}

async function getFastProvider() {
  if (typeof ethers === "undefined") throw new Error("Web3 组件尚未加载完成，请刷新页面。");
  const tasks = BSC_RPCS.map(async (url) => {
    const p = new ethers.JsonRpcProvider(url, { name: "bnb", chainId: 56 }, { staticNetwork: true });
    await p.getBlockNumber();
    return p;
  });
  try {
    return await Promise.any(tasks);
  } catch (e) {
    return new ethers.JsonRpcProvider(BSC_RPCS[0], { name: "bnb", chainId: 56 }, { staticNetwork: true });
  }
}

async function ensureBscNetwork(eth) {
  try {
    await eth.request({ method: "wallet_switchEthereumChain", params: [{ chainId: "0x38" }] });
  } catch (swErr) {
    if (swErr.code === 4902) {
      await eth.request({
        method: "wallet_addEthereumChain",
        params: [{
          chainId: "0x38",
          chainName: "BNB Smart Chain",
          nativeCurrency: { name: "BNB", symbol: "BNB", decimals: 18 },
          rpcUrls: ["https://bsc-dataseed.binance.org"],
          blockExplorerUrls: ["https://bscscan.com"]
        }]
      });
    } else {
      throw swErr;
    }
  }
}

async function connectMainWallet(silent = false) {
  const eth = getWalletProvider();
  if (!eth) {
    toast("未检测到 Web3 钱包，请在 Binance Wallet / MetaMask / OKX 浏览器中打开此页面。", "error");
    return false;
  }
  const btn = document.getElementById("cBtn");
  try {
    btn.disabled = true;
    btn.innerText = "正在连接主钱包...";
    let provider = new ethers.BrowserProvider(eth);
    const net = await provider.getNetwork();
    if (Number(net.chainId) !== 56) {
      await ensureBscNetwork(eth);
      provider = new ethers.BrowserProvider(eth);
    }
    signer = await provider.getSigner();
    mainAddr = await signer.getAddress();
    btn.innerText = "✅ 已连接主钱包: " + shortAddr(mainAddr);
    if (!silent) {
      toast("主钱包已连接 (" + shortAddr(mainAddr) + ")，现在可以代为激活 Agent 或打赏榜单成员！", "success");
    }
    await loadAllChainData();
    return true;
  } catch (e) {
    btn.innerText = "1. 连接普通主钱包 (用于支付 Gas / 打赏)";
    toast("连接取消或切网失败：" + (e.shortMessage || e.message), "error");
    return false;
  } finally {
    btn.disabled = false;
  }
}

async function inspectAgent() {
  const raw = document.getElementById("agentIn").value.trim();
  if (!ethers.isAddress(raw)) {
    return toast("请输入有效的 0x 开头 42 位 Agentic Wallet 地址。", "error");
  }
  const addr = ethers.getAddress(raw);
  const btn = document.getElementById("qBtn");
  try {
    btn.disabled = true;
    btn.innerText = "核验中";
    const provider = await getFastProvider();
    const agentContract = new ethers.Contract(AGENT_ID_ADDR, AGENT_ABI, provider);
    const hub = new ethers.Contract(HUB_V2_ADDR, HUB_V2_ABI, provider);

    const [bal, p] = await Promise.all([
      agentContract.balanceOf(addr),
      hub.profiles(addr)
    ]);

    if (bal === 0n) {
      document.getElementById("agentCard").style.display = "none";
      return toast("地址 (" + shortAddr(addr) + ") 暂未持有币安官方 Agent ID，无法激活。", "error");
    }

    targetAgentAddr = addr;
    const card = document.getElementById("agentCard");
    card.style.display = "block";

    const seed = parseInt(addr.slice(-4), 16);
    document.getElementById("ava").innerText = ICONS[seed % ICONS.length];
    document.getElementById("aDisp").innerText = shortAddr(addr);
    document.getElementById("balDisp").innerText = bal.toString();
    document.getElementById("chkDisp").innerText = p.checkIns.toString();
    document.getElementById("tipDisp").innerText = ethers.formatEther(p.tipsReceived);

    if (p.registered) {
      document.getElementById("statusBadge").innerText = "ACTIVE ON-CHAIN AGENT ✔";
      document.getElementById("roleTitle").innerText = p.roleTag;
      document.getElementById("opDisp").innerText = shortAddr(p.operator);
      document.getElementById("tagIn").value = p.roleTag;
      updateByteCount();
      document.getElementById("actBtn").innerText = "✏️ 更新 Agent 角色定位";

      const chkBtn = document.getElementById("chkBtn");
      chkBtn.style.display = "block";
      const nowSec = Math.floor(Date.now() / 1000);
      const nextTime = Number(p.lastCheckIn) + 43200;
      if (nowSec < nextTime) {
        const left = nextTime - nowSec;
        const hrs = Math.floor(left / 3600);
        const mins = Math.ceil((left % 3600) / 60);
        chkBtn.disabled = true;
        chkBtn.innerText = `⏳ 打卡冷却中 (还剩 ${hrs} 小时 ${mins} 分钟)`;
      } else {
        chkBtn.disabled = false;
        chkBtn.innerText = "📅 代为执行每日链上打卡 (+1 声誉)";
      }
    } else {
      document.getElementById("statusBadge").innerText = "ELIGIBLE AGENT ID HOLDER";
      document.getElementById("roleTitle").innerText = "持有 Agent ID · 尚未激活声誉";
      document.getElementById("opDisp").innerText = "激活时自动绑定当前主钱包";
      document.getElementById("actBtn").innerText = "⚡ 主钱包代为激活此 Agent";
      document.getElementById("chkBtn").style.display = "none";
    }
    toast("已核验到有效 Agent ID 持有地址 (" + shortAddr(addr) + ")！", "success");
  } catch (e) {
    toast("链上核验失败：" + (e.shortMessage || e.message), "error");
  } finally {
    btn.disabled = false;
    btn.innerText = "核验";
  }
}

async function activateForAgent() {
  if (!targetAgentAddr) return toast("请先输入并核验 Agentic Wallet 地址。", "error");
  if (!signer) {
    const ok = await connectMainWallet(true);
    if (!ok) return;
  }
  const tag = document.getElementById("tagIn").value.trim();
  const byteLen = new TextEncoder().encode(tag).length;
  if (!tag) return toast("请输入 Agent 角色定位标签（如 DeFi Alpha Hunter）。", "error");
  if (byteLen > 32) return toast("定位标签过长（当前 " + byteLen + " 字节，上限 32 字节），请精简字数。", "error");

  const btn = document.getElementById("actBtn");
  try {
    btn.disabled = true;
    btn.innerText = "请在主钱包确认上链...";
    toast("正在唤起主钱包确认激活交易...", "info");
    const hub = new ethers.Contract(HUB_V2_ADDR, HUB_V2_ABI, signer);
    const tx = await hub.activateForAgent(targetAgentAddr, tag);
    btn.innerText = "交易打包确认中...";
    toast("交易已发送，等待 BSC 主网打包确认...", "info");
    await tx.wait();
    showTxBanner(tx.hash);
    toast("🎉 激活成功！该 Agent 已正式登记上链并登上全网榜单。", "success");
    await inspectAgent();
    await loadAllChainData();
  } catch (e) {
    toast("激活失败：" + (e.shortMessage || e.reason || e.message), "error");
  } finally {
    btn.disabled = false;
    btn.innerText = "⚡ 主钱包代为激活 / 更新身份";
  }
}

async function checkInForAgent() {
  if (!targetAgentAddr) return;
  if (!signer) {
    const ok = await connectMainWallet(true);
    if (!ok) return;
  }
  const btn = document.getElementById("chkBtn");
  try {
    btn.disabled = true;
    btn.innerText = "请在钱包确认打卡...";
    const hub = new ethers.Contract(HUB_V2_ADDR, HUB_V2_ABI, signer);
    const tx = await hub.checkInForAgent(targetAgentAddr);
    btn.innerText = "打卡上链中...";
    await tx.wait();
    showTxBanner(tx.hash);
    toast("✅ 每日链上打卡成功，声誉积分 +1！", "success");
    await inspectAgent();
    await loadAllChainData();
  } catch (e) {
    toast("打卡失败：" + (e.shortMessage || e.reason || e.message), "error");
    btn.disabled = false;
    btn.innerText = "📅 代为执行每日链上打卡 (+1 声誉)";
  }
}

async function tipAgent(toAgentAddr, btnEl) {
  if (!signer) {
    toast("正在唤起钱包连接以完成打赏...", "info");
    const ok = await connectMainWallet(true);
    if (!ok) return;
  }
  if (mainAddr && toAgentAddr.toLowerCase() === mainAddr.toLowerCase()) {
    return toast("合约安全限制：不能用同一个地址给自己打赏，请切换另一个钱包打赏。", "error");
  }
  const oldText = btnEl ? btnEl.innerText : "打赏";
  try {
    if (btnEl) {
      btnEl.disabled = true;
      btnEl.innerText = "打赏中...";
    }
    toast("请在钱包确认向 " + shortAddr(toAgentAddr) + " 打赏 0.0005 BNB...", "info");
    const hub = new ethers.Contract(HUB_V2_ADDR, HUB_V2_ABI, signer);
    const tx = await hub.tipAgent(toAgentAddr, { value: ethers.parseEther("0.0005") });
    toast("打赏交易已发出，等待链上确认...", "info");
    await tx.wait();
    showTxBanner(tx.hash);
    toast("💸 打赏成功！0.0005 BNB 已点对点直达对方 Agent 钱包。", "success");
    await loadAllChainData();
    if (targetAgentAddr && targetAgentAddr.toLowerCase() === toAgentAddr.toLowerCase()) {
      await inspectAgent();
    }
  } catch (e) {
    toast("打赏未完成：" + (e.shortMessage || e.reason || e.message), "error");
  } finally {
    if (btnEl) {
      btnEl.disabled = false;
      btnEl.innerText = oldText;
    }
  }
}

async function loadAllChainData() {
  try {
    const provider = await getFastProvider();
    const hub = new ethers.Contract(HUB_V2_ADDR, HUB_V2_ABI, provider);

    const total = await hub.getTotalAgents();
    document.getElementById("totalCnt").innerText = total.toString();

    const [addrs, data] = await hub.getRecentAgents();
    const board = document.getElementById("board");
    if (addrs.length === 0) {
      board.innerHTML = '<div style="text-align:center;color:#5e6673;font-size:12px;padding:12px">暂无已激活 Agent，输入 Agentic Wallet 成为第一位！</div>';
    } else {
      board.innerHTML = addrs.map((addr, i) => {
        const p = data[i];
        return `
          <div class="item">
            <div class="item-info">
              <div class="item-tag">${escapeHtml(p.roleTag)}</div>
              <div class="item-addr">${shortAddr(addr)} · 打卡 ${p.checkIns} 次 · 获赏 ${ethers.formatEther(p.tipsReceived)} BNB</div>
            </div>
            <button type="button" class="btn-tip" onclick="tipAgent('${addr}', this)">打赏</button>
          </div>
        `;
      }).join("");
    }

    await loadRecentTxLogs(provider, hub);
  } catch (e) {
    document.getElementById("board").innerHTML = '<div style="text-align:center;color:#848e9c;font-size:12px;padding:12px">榜单同步中，请稍后刷新页面...</div>';
  }
}

async function loadRecentTxLogs(provider, hub) {
  const txList = document.getElementById("txList");
  try {
    const latestBlock = await provider.getBlockNumber();
    const fromBlock = Math.max(0, latestBlock - 1500);

    const [actLogs, chkLogs, tipLogs] = await Promise.all([
      hub.queryFilter(hub.filters.AgentActivated(), fromBlock, latestBlock).catch(() => []),
      hub.queryFilter(hub.filters.CheckedIn(), fromBlock, latestBlock).catch(() => []),
      hub.queryFilter(hub.filters.AgentTipped(), fromBlock, latestBlock).catch(() => [])
    ]);

    const combined = [];
    actLogs.forEach(ev => {
      combined.push({
        block: ev.blockNumber,
        idx: ev.index || 0,
        hash: ev.transactionHash,
        title: `⚡ 激活 [${escapeHtml(ev.args.roleTag)}]`,
        sub: `Agent: ${shortAddr(ev.args.agentWallet)} · 区块 #${ev.blockNumber}`
      });
    });
    chkLogs.forEach(ev => {
      combined.push({
        block: ev.blockNumber,
        idx: ev.index || 0,
        hash: ev.transactionHash,
        title: `📅 链上打卡 (累计 ${ev.args.totalCheckIns} 次)`,
        sub: `Agent: ${shortAddr(ev.args.agentWallet)} · 区块 #${ev.blockNumber}`
      });
    });
    tipLogs.forEach(ev => {
      combined.push({
        block: ev.blockNumber,
        idx: ev.index || 0,
        hash: ev.transactionHash,
        title: `💸 打赏 ${ethers.formatEther(ev.args.amount)} BNB`,
        sub: `${shortAddr(ev.args.from)} → ${shortAddr(ev.args.toAgent)}`
      });
    });

    combined.sort((a, b) => (b.block - a.block) || (b.idx - a.idx));
    const recent = combined.slice(0, 5);

    if (recent.length === 0) {
      txList.innerHTML = '<div style="text-align:center;color:#5e6673;font-size:12px;padding:12px">近 1 小时暂无新交易，发起激活或打赏后将实时显示于此。</div>';
      return;
    }

    txList.innerHTML = recent.map(item => `
      <div class="item">
        <div class="item-info">
          <div class="item-tag" style="color:#eaecef">${item.title}</div>
          <div class="item-addr">${item.sub}</div>
        </div>
        <a class="tx-link" href="https://bscscan.com/tx/${item.hash}" target="_blank">查看 Tx ↗</a>
      </div>
    `).join("");
  } catch (e) {
    txList.innerHTML = '<div style="text-align:center;color:#848e9c;font-size:12px;padding:12px">点击右上角“全部交易 ↗”前往 BscScan 查验历史记录。</div>';
  }
}
